name: Gobi compiled Go lifecycle

on:
  push:
  pull_request:

jobs:
  compiled-lifecycle:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-go@v5
        with:
          go-version: 1.24.x

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: |
            package-lock.json
            interaction/package-lock.json
            compiled-frontend/package-lock.json

      - name: Install locked JavaScript dependencies
        run: |
          npm ci --no-audit --no-fund
          npm ci --prefix interaction --no-audit --no-fund
          npm ci --prefix compiled-frontend --no-audit --no-fund

      - name: Install Playwright Chromium
        run: interaction/node_modules/.bin/playwright install --with-deps chromium

      - name: Run inherited gates and compiled lifecycle matrix
        env:
          GONDOLA_SKIP_NPM_CI: "1"
          GONDOLA_SKIP_BROWSER_INSTALL: "1"
        run: bash run-compiled-lifecycle.sh

      - name: Preserve build, matrix, demo, and inherited evidence
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: gobi-compiled-lifecycle-artifacts
          path: |
            compiled-lifecycle/builds/attempts/**
          compiled-lifecycle/builds/feasibility/**
            compiled-lifecycle/coordination/reports/E-BUILD.md
            compiled-lifecycle/coordination/receipts/E-BUILD.json
            compiled-lifecycle/conformance/evidence/**
            compiled-lifecycle/demo/evidence/**
            compiled-methods/artifacts/attempts/**
            compiled-methods/coordination/reports/E-DEMO.md
            compiled-methods/coordination/receipts/E-DEMO.json
            compiled-backends/artifacts/**
            compiled-backends/coordination/reports/E-DEMO.md
            compiled-backends/coordination/receipts/E-DEMO.json
            compiled-frontend/artifacts/**
          if-no-files-found: warn
