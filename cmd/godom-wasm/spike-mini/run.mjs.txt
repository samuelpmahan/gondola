import { createServer } from 'node:http';
import { readFileSync } from 'node:fs';
import { createRequire } from 'node:module';
const root='/tmp/godom-spike';
const server=createServer((req,res)=>{const name=req.url==='/'?'index.html':req.url.slice(1); res.setHeader('Content-Type',name.endsWith('.wasm')?'application/wasm':name.endsWith('.js')?'text/javascript':'text/html'); res.end(readFileSync(`${root}/${name}`));});
await new Promise(resolve=>server.listen(0,'127.0.0.1',resolve));
const port=server.address().port;
const require=createRequire(import.meta.url); const {chromium}=require('/workspace/scratch/39a56936f173/gondola/interaction/node_modules/playwright');
const browser=await chromium.launch({headless:true,executablePath:'/tmp/chromium',args:['--no-sandbox']});
const page=await browser.newPage(); page.on('console',m=>console.log('console',m.type(),m.text())); page.on('pageerror',e=>console.log('pageerror',e.stack));
await page.goto(`http://127.0.0.1:${port}/`); await page.waitForTimeout(1000);
console.log('SPIKE_RESULT',await page.locator('body').getAttribute('data-result'),await page.locator('#root').textContent());
await browser.close(); server.close();
